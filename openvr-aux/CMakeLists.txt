cmake_minimum_required(VERSION 3.21)
project(qptp LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "qptp requires 64-bit Windows")
endif()
add_library(qptp_minhook STATIC vendor/minhook/src/buffer.c vendor/minhook/src/hook.c vendor/minhook/src/trampoline.c vendor/minhook/src/hde/hde64.c)
target_include_directories(qptp_minhook PUBLIC vendor/minhook/include)
target_compile_definitions(qptp_minhook PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(driver_qptp SHARED src/driver.cpp src/debug_udp.cpp src/fusion.cpp src/skeletal_driver.cpp src/skeletal_mapping.cpp vendor/hand_simulation/hand_simulation.cpp)
target_compile_features(driver_qptp PRIVATE cxx_std_17)
if(MSVC)
  target_link_options(driver_qptp PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/driver_qptp.map")
endif()
target_compile_definitions(driver_qptp PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(driver_qptp PRIVATE vendor/openvr vendor/hand_simulation)
target_link_libraries(driver_qptp PRIVATE ws2_32 qptp_minhook)
set_target_properties(driver_qptp PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/driver_qptp/bin/win64"
  RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_CURRENT_SOURCE_DIR}/driver_qptp/bin/win64"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/driver_qptp/bin/win64")

include(CTest)
if(BUILD_TESTING)
  add_executable(skeleton_test tests/skeleton_test.cpp src/skeletal_driver.cpp src/skeletal_mapping.cpp vendor/hand_simulation/hand_simulation.cpp)
  target_compile_features(skeleton_test PRIVATE cxx_std_17)
  target_compile_definitions(skeleton_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_include_directories(skeleton_test PRIVATE src vendor/openvr vendor/hand_simulation)
  if(MSVC)
    target_compile_options(skeleton_test PRIVATE /Od)
  endif()
  target_link_libraries(skeleton_test PRIVATE qptp_minhook)
  add_test(NAME skeleton COMMAND skeleton_test)
  set_tests_properties(skeleton PROPERTIES TIMEOUT 10)
  add_executable(udp_test tests/udp_test.cpp src/debug_udp.cpp)
  target_compile_features(udp_test PRIVATE cxx_std_17)
  target_compile_definitions(udp_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_include_directories(udp_test PRIVATE src)
  target_link_libraries(udp_test PRIVATE ws2_32)
  add_test(NAME udp COMMAND udp_test)
endif()

# Manual hardware probe; not a CTest because it needs a running headset/SteamVR.
add_executable(skeleton_check tests/skeleton_check.cpp)
target_compile_features(skeleton_check PRIVATE cxx_std_17)
target_compile_definitions(skeleton_check PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(skeleton_check PRIVATE vendor/openvr)
foreach(resource skeleton_check_actions.json skeleton_check_bindings.json)
  add_custom_command(TARGET skeleton_check POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/${resource} $<TARGET_FILE_DIR:skeleton_check>/${resource})
endforeach()
